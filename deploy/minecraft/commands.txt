execute in minecraft:spawn run gamerule doMobSpawning false

execute in minecraft:spawn run gamerule doTraderSpawning false
execute in minecraft:spawn run gamerule doInsomnia false
execute in minecraft:spawn run gamerule doWeatherCycle false
execute in minecraft:spawn run gamerule spawnRadius 0

# creeper ni de feu dans les salles.

lp group roi permission set essentials.warps.roi true
